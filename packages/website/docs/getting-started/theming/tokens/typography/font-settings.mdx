import { Example } from '@site/src/components';

import { FontDefaultUnitsPreview } from './preview_font_settings';

<Example>
  <Example.Description>
    ### `euiTheme.font.defaultUnits` <Badge color="hollow">token</Badge>

    Sets the default units used for font size & line height set by UI components like EuiText or EuiTitle. Defaults to `rem`.

    Note: This may override by some internal usages, e.g. EuiText's `relative` size which must use `em`.

  </Example.Description>
  <Example.Preview>
    <FontDefaultUnitsPreview />
  </Example.Preview>
  <Example.Snippet>
    ```tsx
    <EuiProvider modify={{ font: { defaultUnits: 'px' } }}>
      <EuiText>
        <p>Hello world</p>
      </EuiText>
    </EuiProvider>
    ```
  </Example.Snippet>
</Example>

## Title

These tokens set the defaults for titles and headings. Prefer [EuiTitle](../../../../components/display/title.mdx) when rendering titles rather than applying the tokens directly.

import { FontTitleWeightPreview } from './preview_font_settings';

<Example>
  <Example.Description>
    ### `euiTheme.font.title.weight` <Badge color="hollow">token</Badge>

    A font weight key for setting the base weight for titles and headings. Maps to a value from [`euiTheme.font.weight`](./font-weight.mdx).

  </Example.Description>
  <Example.Preview>
    <FontTitleWeightPreview />
  </Example.Preview>
  <Example.Snippet>
    ```tsx
    css`
      font-weight: ${euiTheme.font.weight[euiTheme.font.title.weight]};
    `
    ```
  </Example.Snippet>
</Example>
